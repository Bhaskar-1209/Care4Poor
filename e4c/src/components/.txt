import React, { useState, useEffect } from "react";
import { Link } from "react-router-dom";

const Navbar = () => {
  const [isOpen, setIsOpen] = useState(false);
    const [scrolled, setScrolled] = useState(false);

  useEffect(() => {
    const handleScroll = () => setScrolled(window.scrollY > 50);
    window.addEventListener("scroll", handleScroll);
    return () => window.removeEventListener("scroll", handleScroll);
  }, []);

  return (
    <div>
      <nav className={`fixed top-10 left-1 w-full flex justify-between items-center bg-yellow-300 p-4 rounded-full shadow-lg z-50 mx-auto max-w-[95%] mt-4
      {
          scrolled
            ? "bg-black/90 border-black text-white py-3"
            : "bg-yellow-400 border-black text-black py-4"
        }`}>
        {/* Logo + Title */}
        <div className="flex items-center space-x-4">
          <img src="/logo.png" alt="Logo" className="w-10 h-10 object-contain" />
          <span
            className={`font-bold text-lg ${
              scrolled ? "text-green-300" : "text-black"
            }`}
          >
            CharityForGood
          </span>
        </div>

        {/* Desktop Menu */}
        <ul className="hidden md:flex space-x-6 font-semibold text-black">
          <li><Link to="/">Home</Link></li>
          <li><Link to="/about">About Us</Link></li>
          <li><Link to="/initiatives">Our Initiatives</Link></li>
          <li><Link to="/projects">Projects</Link></li>
          <li><Link to="/join">Join Us</Link></li>
          <li><Link to="/contact">Contact Us</Link></li>
        </ul>

        {/* Mobile Menu Button */}
        <button
          onClick={() => setIsOpen(true)}
          className="md:hidden text-black text-2xl"
          aria-label="Open menu"
        >
          ☰
        </button>

        {/* Fullscreen Mobile Menu with Slide Animation */}
        <div
          className={`fixed inset-0 bg-yellow-300 flex flex-col items-center justify-center space-y-8 text-black font-semibold text-2xl z-50 transform transition-transform duration-300 ease-in-out ${
            isOpen ? "translate-x-0" : "translate-x-full"
          }`}
        >
          {/* Close Button */}
          <button
            onClick={() => setIsOpen(false)}
            className="absolute top-6 right-6 text-4xl font-bold"
            aria-label="Close menu"
          >
            ✕
          </button>

          {/* Menu Links */}
          <Link to="/" onClick={() => setIsOpen(false)}>Home</Link>
          <Link to="/about" onClick={() => setIsOpen(false)}>About Us</Link>
          <Link to="/initiatives" onClick={() => setIsOpen(false)}>Our Initiatives</Link>
          <Link to="/projects" onClick={() => setIsOpen(false)}>Projects</Link>
          <Link to="/join" onClick={() => setIsOpen(false)}>Join Us</Link>
          <Link to="/contact" onClick={() => setIsOpen(false)}>Contact Us</Link>
        </div>
      </nav>

      {/* Spacer to prevent content from hiding behind fixed navbar */}
      <div className="h-20"></div>
    </div>
  );
};

export default Navbar;
